def gh(endpoint=None,query=None):
 args=['gh','api',endpoint] if endpoint else ['gh','api','graphql','--input','-']
 data=None if endpoint else json.dumps({'query':query})
 r=subprocess.run(args,input=data,text=True,capture_output=True,timeout=90)
 if r.returncode: raise RuntimeError(r.stderr[:350])
 result=json.loads(r.stdout)
 if isinstance(result,dict) and result.get('errors'):raise RuntimeError(str(result['errors'])[:350])
 return result
